my-nuxt-app/

├── components/             # Vue компоненты
├── layouts/                # Шаблоны для страниц

├── pages/
│   ├── blog/               # Блог

├── static/                 # Статические файлы, не обрабатываемые Webpack
├── store/                  # Vuex хранилище для управления состоянием
├── nuxt.config.js          # Конфигурационный файл Nuxt.js
├── package.json            # Файл зависимостей и скриптов
└── README.md               # Документация проекта
